Неявное приведение - компилятор сам решает к какому типу надо привести значение

Явное приведение - четко определяется разработчиком в какой тип приводить переменную

```
float x = 6.5F; //если не написать F то значение примет вид double
int y = 5;
int a = x+y; //будет ошибка так как неявное приведение типов невозможно
```

Разрешенные неявные преобразования


| Из типа |                            К типу                             |
| ------- | :-----------------------------------------------------------: |
| byte    | short, ushort, int, uint, long, ulong, float, double, decimal |
| sbyte   |           short, int, long, float, double, decimal            |
| short   |               int, long, float, double, decimal               |
| ushort  |        int, uint, long, ulong, float, double, decimal         |
| int     |                 long, float, double, decimal                  |
| uint    |              long, ulong, float, double, decimal              |
| long    |                    float, double, decimal                     |
| ulong   |                    float, double, decimal                     |
| char    |     ushort, int uint, long, ulong, float, double, decimal     |
| float   |                            double                             |
